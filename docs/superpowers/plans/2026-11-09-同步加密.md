# 同步目录端到端加密（Spec 14）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 `push`/`pull` 增加可选的端到端加密：推送时把 zip 封成 AES-256-GCM 信封 `.spk` 落同步目录，接收方 `pull` 时口令解密后照常导入，网盘侧只见密文。

**Architecture:** 新增 L0 纯函数层 `sessionpack/crypto.py`（全项目唯一允许 `import cryptography` 的文件，由 lint 新增 `E_CRYPTO` 规则守卫），负责 PBKDF2 派生密钥与 AES-GCM 信封编解码；`sync.py` 增加信封模式（扫目录同时收 `*.zip` 与 `*.spk`、按 `archive_id` 判重、解密只在内存进行、原子字节落盘）；口令采集只在 CLI 层（环境变量 `SESSIONPACK_PASSPHRASE` 优先，交互 `getpass` 回退），core 层不弹窗。

**Tech Stack:** Python 3.11+，标准库（`hashlib`/`secrets`/`getpass`/`zipfile`/`io`/`json`/`os`/`uuid`）+ 运行时依赖 `cryptography>=42`（AESGCM、PBKDF2HMAC、InvalidTag）。

```python
"""同步目录端到端加密（Spec 14 D1-D4）：全项目唯一允许 import cryptography 的文件。

信封格式（版本 0x01，头长 38B）：
    magic "SPKE1"(5B) | 版本 0x01(1B) | salt(16B) | iterations(uint32 大端 4B)
    | nonce(12B) | ciphertext(含 AESGCM 16B 认证 tag)
KDF 为 PBKDF2-HMAC-SHA256；iterations 写入信封头，日后升档旧信封仍可解。
AAD 取信封文件名的字节：防止密文在文件之间整体搬运、冒名顶替。
本层是纯函数，不弹窗、不读环境变量、不碰磁盘——口令一律由调用方传入。
"""

from __future__ import annotations

import os
import secrets

from cryptography.exceptions import InvalidTag
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC

MAGIC = b"SPKE1"
VERSION = 1
KDF_SALT_LEN = 16
NONCE_LEN = 12
PBKDF2_ITERATIONS = 600_000
_HEADER_LEN = len(MAGIC) + 1 + KDF_SALT_LEN + 4 + NONCE_LEN   # 38

NAME_PREFIX = "spk-"
NAME_SUFFIX = ".spk"


class EnvelopeError(ValueError):
    """信封非法：magic/版本/长度不对、口令错误或认证失败。消息含错误码不含明文。"""


def derive_key(passphrase: str, salt: bytes, iterations: int) -> bytes:
    """PBKDF2-HMAC-SHA256 派生 32 字节密钥；iterations 由调用方从信封头读出。"""
    kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=32, salt=salt,
                     iterations=iterations)
    return kdf.derive(passphrase.encode("utf-8"))


def new_envelope_name() -> str:
    """随机信封名（与内容无关）：spk-<32hex>.spk。"""
    return f"{NAME_PREFIX}{secrets.token_hex(16)}{NAME_SUFFIX}"


def is_envelope_name(name: str) -> bool:
    """宽松判定：前缀 spk- 且后缀 .spk 即视为信封（结构问题留给解密阶段）。"""
    return name.startswith(NAME_PREFIX) and name.endswith(NAME_SUFFIX)


def encrypt_envelope(data: bytes, passphrase: str, envelope_name: str) -> bytes:
    """加密为信封字节流；envelope_name 的字节作 AAD 参与认证。"""
    salt = os.urandom(KDF_SALT_LEN)
    key = derive_key(passphrase, salt, PBKDF2_ITERATIONS)
    nonce = os.urandom(NONCE_LEN)
    ct = AESGCM(key).encrypt(nonce, data, envelope_name.encode("utf-8"))
    return (MAGIC + bytes([VERSION]) + salt
            + PBKDF2_ITERATIONS.to_bytes(4, "big") + nonce + ct)


def decrypt_envelope(blob: bytes, passphrase: str, envelope_name: str) -> bytes:
    """解密信封字节流；一切失败统一抛 EnvelopeError（E202 结构 / E201 口令）。"""
    if len(blob) < _HEADER_LEN:
        raise EnvelopeError("E202: 信封长度不足")
    if blob[:len(MAGIC)] != MAGIC:
        raise EnvelopeError("E202: magic 不匹配")
    if blob[len(MAGIC)] != VERSION:
        raise EnvelopeError(f"E202: 不支持的信封版本 {blob[len(MAGIC)]}")
    salt = blob[6:6 + KDF_SALT_LEN]
    iterations = int.from_bytes(blob[22:26], "big")
    nonce = blob[26:_HEADER_LEN]
    ct = blob[_HEADER_LEN:]
    key = derive_key(passphrase, salt, iterations)
    try:
        return AESGCM(key).decrypt(nonce, ct, envelope_name.encode("utf-8"))
    except InvalidTag as exc:
        raise EnvelopeError("E201: 口令错误或信封被篡改") from exc
```

```python
def sync_encrypt(root: Path) -> bool:
    """同步目录加密开关；缺失、缺段或值非 bool 都返回 False（G4：不抛异常）。"""
    data, _ = load_config(root)
    section = data.get("sync")
    if not isinstance(section, dict):
        return False
    value = section.get("encrypt")
    return value is True
```

```python
def set_sync_encrypt(root: Path, enabled: bool) -> None:
    """把 `[sync].encrypt` 设为布尔值；fail-closed 语义与 set_sync_dir 相同。"""
    root = Path(root)
    root.mkdir(parents=True, exist_ok=True)
    data, fatal = _read_toml(root)
    if fatal:
        raise ConfigWriteError(
            "config.toml 解析失败，已拒绝写入以免丢失现有内容；"
            "请修复语法后重试，或手工维护 [sync] 段的 encrypt 键")
    data = dict(data)
    section = data.get("sync")
    section = dict(section) if isinstance(section, dict) else {}
    section["encrypt"] = bool(enabled)
    data["sync"] = section
    problems = _unfaithful(data)
    if problems:
        raise ConfigWriteError(
            "config.toml 含无法保真保留的内容：" + "、".join(problems)
            + "；已拒绝写入以免损坏你的文件，请手工维护 [sync] 段的 encrypt 键")
    path = root / CONFIG_NAME
    tmp = root / (CONFIG_NAME + ".tmp")
    tmp.write_text(_dump_toml(data), encoding="utf-8")
    try:
        os.replace(tmp, path)
    except OSError:
        tmp.unlink(missing_ok=True)
        raise
```

```python
def _is_envelope_name(name: str) -> bool:
    return name.startswith(crypto.NAME_PREFIX) and name.endswith(crypto.NAME_SUFFIX)


def _scan_sync_candidates(directory: Path) -> list[Path]:
    """同步目录候选：`*.zip` + `*.spk`，各自排序；隐藏名与临时前缀不入候选。

    本机 archives/ 不用本函数（仍由 _scan_zips 只认 zip，spec §4 D7）。
    """
    directory = Path(directory)
    if not directory.is_dir():
        return []
    out: list[Path] = []
    for pattern in ("*.zip", f"*{crypto.NAME_SUFFIX}"):
        for p in sorted(directory.glob(pattern)):
            if p.name.startswith(".") or p.name.startswith(_TEMP_PREFIXES):
                continue
            out.append(p)
    return out


def _envelope_archive_id(path: Path, passphrase: str) -> str:
    """解出信封内 zip 的 archive_id；全程只在内存，不落盘。"""
    data = crypto.decrypt_envelope(path.read_bytes(), passphrase, path.name)
    with zipfile.ZipFile(io.BytesIO(data)) as zf:
        manifest = json.loads(zf.read(archive.MANIFEST_NAME))
    return manifest["archive_id"]


def _atomic_write(tmp: Path, blob: bytes, dest: Path) -> None:
    """字节级原子落盘：确保父目录存在，写临时名再 os.replace；
    目标已存在视为并发先到，良性跳过（与 _atomic_copy 同口径）。"""
    try:
        tmp.parent.mkdir(parents=True, exist_ok=True)
        tmp.write_bytes(blob)
        os.replace(tmp, dest)
    except (PermissionError, FileExistsError) as exc:
        tmp.unlink(missing_ok=True)
        if dest.exists():
            return
        raise exc
    except OSError:
        tmp.unlink(missing_ok=True)
        raise
```

```python
def push_archives(archives_dir: Path, sync_dir: Path, dry_run: bool = False,
                  *, encrypt: bool = False,
                  passphrase: str | None = None) -> SyncResult:
    """把本地包推进同步目录（只增不删，G7：源目录零改动）。

    encrypt=True 时把每个包封成随机名 .spk 信封落盘，判重靠解出信封内
    zip 的 archive_id（信封名与内容无关，无法按文件名比对）；
    encrypt=False 时行为与历史版本逐字节一致。
    """
    if encrypt and not passphrase:
        raise ValueError("E201: 加密推送需要口令")
    pushed: list[str] = []
    existing = 0
    unverified: list[str] = []
    errors: list[str] = []
    total = 0
    candidates = _scan_sync_candidates(sync_dir)
    remote_names = {p.name for p in candidates}
    have_ids: set[str] = set()
    if encrypt:
        for p in candidates:
            if not _is_envelope_name(p.name):
                continue
            try:
                have_ids.add(_envelope_archive_id(p, passphrase))
            except (crypto.EnvelopeError, OSError, ValueError, KeyError,
                    zipfile.BadZipFile):
                continue          # 无关或损坏信封不参与判重，也不被本命令改动
    for src in _scan_zips(archives_dir):
        if encrypt:
            check = archive.verify_archive(src)
            if not check.ok or check.manifest is None:
                codes = ", ".join(issue.code for issue in check.issues)
                unverified.append(f"{src.name}: {codes or '无 manifest'}")
                continue
            if check.manifest["archive_id"] in have_ids:
                existing += 1
                continue
            try:
                size = src.stat().st_size
            except OSError as exc:
                errors.append(f"{src.name}: {failure_text(exc)}")
                continue
            if not dry_run:
                name = crypto.new_envelope_name()
                try:
                    blob = crypto.encrypt_envelope(src.read_bytes(),
                                                   passphrase, name)
                    tmp = sync_dir / f".tmp-{name}." \
                                     f"{os.getpid()}.{uuid.uuid4().hex[:8]}"
                    _atomic_write(tmp, blob, sync_dir / name)
                except (OSError, crypto.EnvelopeError) as exc:
                    errors.append(f"{src.name}: {failure_text(exc)}")
                    continue
            pushed.append(src.name)
            total += size
            continue
        if src.name in remote_names:
            existing += 1
            continue
        check = archive.verify_archive(src)
        if not check.ok:
            codes = ", ".join(issue.code for issue in check.issues)
            unverified.append(f"{src.name}: {codes}")
            continue
        try:
            size = src.stat().st_size
        except OSError as exc:
            errors.append(f"{src.name}: {failure_text(exc)}")
            continue
        if not dry_run:
            try:
                _atomic_copy(src, Path(sync_dir) / src.name)
            except OSError as exc:
                errors.append(f"{src.name}: {failure_text(exc)}")
                continue
        pushed.append(src.name)
        total += size
    return SyncResult(pushed=len(pushed), skipped_existing=existing,
                      skipped_unverified=tuple(unverified),
                      bytes=total, errors=tuple(errors))
```

说明（评审时容易误改的点，勿"顺手优化"）：

- 累加器风格（局部 list + 末尾构造 `SyncResult`）与既有实现一致——`SyncResult` 的 `skipped_unverified`/`errors` 是 tuple，不能 append。
- 不预先 `mkdir` 两个目录：明文模式历史上从不建 `archives_dir`，且 `dry_run` 不得创建 `sync_dir`（既有 `test_push_dry_run_writes_nothing` 断言 `not share.exists()`）。密文模式的目录创建由 `_atomic_write` 内部 `tmp.parent.mkdir` 完成（只在非 dry-run 落盘时触发），与明文分支由 `_atomic_copy` 内部 `dest.parent.mkdir` 的行为对齐。
- 密文分支报告字节沿用"明文 zip 的 st_size"（与明文分支同一口径），信封略大于 zip 属正常。
- 明文分支逐行保持原逻辑：verify → stat → `_atomic_copy`，错误一律 `failure_text(exc)` 脱敏。

```python
                  *, encrypt: bool = False,
                  passphrase: str | None = None) -> SyncResult:
    """把同步目录的包拉进本机 archives/ 并导入索引（只增不删，G7：源零改动）。

    `*.zip` 分支与历史版本逐字节一致；encrypt=True 时另解 `*.spk` 信封：

    errors: list[str] = []
    sessions_added = 0
    messages_added = 0

    local_names = {p.name for p in _scan_zips(archives_dir)}

            try:
                _atomic_copy(src, dest)
            except OSError as exc:
                errors.append(f"{src.name}: {failure_text(exc)}")
                continue
            try:
                imported = archive.import_archive(conn, dest)
            except (archive.ArchiveError, OSError, sqlite3.Error) as exc:
                errors.append(f"{src.name}: 已复制但导入失败 {failure_text(exc)}")
                continue
            pulled.append(src.name)
            sessions_added += imported.sessions_created
            messages_added += imported.messages_inserted
            total += size
            if tail:
                local_ids.add(tail)
            local_names.add(src.name)
            continue
        if not encrypt:
            continue            # 不开加密时信封视为不可见（零行为变更）
        try:
            data = crypto.decrypt_envelope(src.read_bytes(), passphrase,
                                           src.name)
            with zipfile.ZipFile(io.BytesIO(data)) as zf:
                manifest = json.loads(zf.read(archive.MANIFEST_NAME))
            archive_id = manifest["archive_id"]
            canon = (f"{_slug(manifest['project']['project_id'])}-"
                     f"{manifest['created_at'][:10]}-{archive_id[:12]}.zip")
        except (crypto.EnvelopeError, zipfile.BadZipFile, KeyError, ValueError):
            invalid.append(f"{src.name}: 口令错误或文件损坏")
            continue
        if archive_id[:12] in local_ids or canon in local_names:
            existing += 1
            continue
        dest = archives_dir / canon
        tmp = archives_dir / f".tmp-{canon}.{os.getpid()}." \
                             f"{uuid.uuid4().hex[:8]}"
        try:
            _atomic_write(tmp, data, dest)
        except OSError as exc:
            errors.append(f"{src.name}: {failure_text(exc)}")
            continue
        check = archive.verify_archive(dest)
        if not check.ok:
            dest.unlink(missing_ok=True)
            invalid.append(f"{src.name}: "
                           f"{', '.join(i.code for i in check.issues)}")
            continue
        try:
            imported = archive.import_archive(conn, dest)
        except (archive.ArchiveError, OSError, sqlite3.Error) as exc:
            errors.append(f"{src.name}: 已落盘但导入失败 {failure_text(exc)}")
            continue
        pulled.append(canon)
        sessions_added += imported.sessions_created
        messages_added += imported.messages_inserted
        total += src.stat().st_size
        local_ids.add(archive_id[:12])
        local_names.add(canon)
    return SyncResult(pulled=len(pulled), skipped_existing=existing,
                      skipped_unverified=(), bytes=total,
                      sessions_added=sessions_added,
                      messages_added=messages_added,
                      skipped_invalid=tuple(invalid), errors=tuple(errors))
```

说明（评审时容易误改的点，勿"顺手优化"）：

- zip 分支保持现状判重（尾 12 位 + 文件名）、校验、`_atomic_copy`、导入、错误脱敏全部逐行不动；`local_ids`/`local_names` 的导入后追加逻辑保证 mixed 用例"先到先赢"。
- 信封分支**先解密、内存读 manifest、再落盘**：GCM 通过即证明内容未被篡改且文件名未改（D5），因此无需"先落临时 zip 再 verify"的旧草案写法；`verify_archive(dest)` 只是对本机文件的二次防线，失败必须 `unlink` 撤回，守住"坏包不落本机"的既有语义。
- 规范包名三要素（slug/日期/id 前 12 位）与 `archive/build.py` L166 的 `zip_name` 同一公式，两处若改动必须同步——测试 `test_pull_encrypt_roundtrip` 用 src 构建出的真实包名锁死这一点。
- `total` 在信封分支记信封文件大小（实际传输字节），与明文分支记 zip 大小同为"本报告实际搬运量"口径。

```python
def _passphrase(prompt: str) -> str | None:
    """口令采集（D4）：环境变量优先（脚本/计划任务静默非交互），
    否则 getpass 两次一致。空口令或两次不一致返回 None，由调用方出

    env = os.environ.get("SESSIONPACK_PASSPHRASE")
    if env:
        return env
    first = getpass.getpass(prompt)
    if not first:
        return None
    if getpass.getpass("再次输入确认: ") != first:
        return None
    return first
```

```python
def find_archives_with_session(directory: Path, session_id: str,
                               passphrase: str | None = None) -> list[str]:
    """列出 directory 中含该会话（未 purge 卡）的包文件名。

    明文 zip 逐字节按现状扫；passphrase 非 None 时另解 `*.spk` 信封（D9）。
    解密失败按"不含"跳过——维持 purge 报告"宁漏勿错"取态：报告漏报优于
    把口令错误的无关信封诬报为"仍含旧包"。
    """
    if not directory.is_dir():
        return []
    card = f"sessions/{session_id}/meta.json"
    hits: list[str] = []
    for path in _scan_sync_candidates(directory):
        if _is_envelope_name(path.name):
            if passphrase is None:
                continue
            try:
                data = crypto.decrypt_envelope(path.read_bytes(), passphrase,
                                               path.name)
                with zipfile.ZipFile(io.BytesIO(data)) as zf:
                    if card not in zf.namelist():
                        continue
                    if json.loads(zf.read(card)).get("status") == "purged":
                        continue
            except (crypto.EnvelopeError, OSError, zipfile.BadZipFile, KeyError,
                    ValueError, AttributeError):
                continue
            hits.append(path.name)
            continue
        try:
            with zipfile.ZipFile(path) as zf:
                if card not in zf.namelist():
                    continue
                if json.loads(zf.read(card)).get("status") == "purged":
                    continue
                hits.append(path.name)
        except (OSError, zipfile.BadZipFile, KeyError, ValueError,
                AttributeError):
            continue
    return hits
```

```python
def run_reencrypt(args) -> CommandResult:
    """一次性把同步目录明文 zip 转为加密信封（D8，fail-safe）。

    每个包严格走"verify → 加密写 .tmp-enc-* → 解密自检字节相等 →
    os.replace 改名 → 删原 zip"，自检失败绝不动原包。
    """
    lay = layout(args)
    target = resolve_sync_dir(args, lay)
    if target is None:
        return CommandResult(EXIT_USAGE, err_lines=_missing_dir_errors(lay))
    if not target.is_dir():
        return CommandResult(EXIT_USAGE, err_lines=[f"同步目录不存在: {target}"])
    passphrase = _passphrase("旧明文包加密口令: ")
    if passphrase is None:
        return CommandResult(EXIT_USAGE,
                             err_lines=["已取消：加密需要口令"
                                        "（两次输入为空或不一致）"])
    converted = 0
    skipped: list[str] = []
    errors: list[str] = []
    for src in sync._scan_zips(target):
        check = archive.verify_archive(src)
        if not check.ok:
            codes = ", ".join(issue.code for issue in check.issues)
            skipped.append(f"{src.name}: 校验未通过（{codes}）")
            continue
        name = crypto.new_envelope_name()
        tmp = target / f".tmp-enc-{name}"
        try:
            plain = src.read_bytes()
            blob = crypto.encrypt_envelope(plain, passphrase, name)
            try:
                tmp.write_bytes(blob)
            except OSError as exc:
                errors.append(f"{src.name}: {failure_text(exc)}")
                continue
            if crypto.decrypt_envelope(tmp.read_bytes(), passphrase, name) != plain:
                errors.append(f"{src.name}: 自检失败（解密与原包不一致），已保留原包")
                continue
            try:
                os.replace(tmp, target / name)
            except OSError as exc:
                errors.append(f"{src.name}: {failure_text(exc)}")
                continue
            src.unlink()
            converted += 1
        except (crypto.EnvelopeError, OSError) as exc:
            errors.append(f"{src.name}: {failure_text(exc)}")
        finally:
            tmp.unlink(missing_ok=True)
    payload = {"sync_dir": str(target), "converted": converted,
               "skipped": skipped, "errors": errors}
    lines = [f"已转换 {converted} 个包为加密信封，"
             f"跳过 {len(skipped)}，失败 {len(errors)}"]
    lines += [f"跳过: {item}" for item in skipped]
    lines += [f"转换失败: {item}" for item in errors]
    code = EXIT_ERROR if errors else EXIT_OK
    return CommandResult(code, out_lines=lines, data=payload)


def cmd_reencrypt(args) -> int:
    return emit(run_reencrypt(args), args.json)
```
