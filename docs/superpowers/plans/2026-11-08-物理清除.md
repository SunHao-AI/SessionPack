在 `sessionpack/catalog/write.py` 中，`mark_session_deleted` 之后新增（与 `insert_session_tombstone` 同构，SQL 全部是受管表字面量、本文件在 `SQL_WRITE_ALLOWED` 豁免集内）：

```python
def purge_session(conn, session_id: str, *, now: str) -> bool:
    """物理清除：sessions 行置 purged 终态并清空内容字段，抹掉消息与 FTS。

    仅 deleted 态可进（G2）：True=本次真清除；purged 幂等、active 拒绝均返 False。
    sessions 行保留，作为 purge 卡载体与 install 判家依据。
    与 Spec D2 语句顺序对调：先 UPDATE 判态再抹数据——active 误调时必须
    原样拒掉，不能先把消息删了再发现状态不对。
    """
    cur = conn.execute(
        "UPDATE sessions SET status='purged', title=NULL, summary=NULL, "
        "content_hash=NULL, updated_at=? "
        "WHERE session_id=? AND status='deleted'",
        (now, session_id),
    )
    if cur.rowcount == 0:
        return False
    conn.execute("DELETE FROM message_fts WHERE session_id=?", (session_id,))
    conn.execute("DELETE FROM messages WHERE session_id=?", (session_id,))
    return True


def insert_session_purge_marker(conn, *, session_id: str, project_id: str,
                                source_kind: str, created_at: str,
                                updated_at: str) -> None:
    """归档 purge 卡的强制落卡语义（三道卡②的执行端，与墓碑分支对称）。

    行存在 → 抹消息与 FTS、置 purged、清内容字段（覆盖 active/deleted/purged
    三态）；行不存在（新设备）→ 直接 INSERT 终态卡。
    """
    conn.execute("DELETE FROM message_fts WHERE session_id=?", (session_id,))
    conn.execute("DELETE FROM messages WHERE session_id=?", (session_id,))
    cur = conn.execute(
        "UPDATE sessions SET status='purged', title=NULL, summary=NULL, "
        "content_hash=NULL, updated_at=? WHERE session_id=?",
        (updated_at, session_id),
    )
    if cur.rowcount == 0:
        conn.execute(
            "INSERT INTO sessions(session_id, project_id, source_kind, title, "
            "summary, created_at, updated_at, status) "
            "VALUES(?,?,?,?,?,?,?,'purged')",
            (session_id, project_id, source_kind, None, None,
             created_at, updated_at),
        )
```

（语义说明：这里不用 INSERT OR IGNORE 弱语义——本机 active/deleted 态收到 purge 包时必须强制抹除升级，否则"已导入未收卡"的设备会泄漏正文；purge 传播语义强于墓碑。CLI 的"仅 deleted 可 purge"门槛由 `purge_session` 单独保证，两者不冲突。对称参照：墓碑分支 `mark_session_deleted` 失败才 `insert_session_tombstone`，即墓碑对已存在的行也是 UPDATE 强制降级。）

两处修改：

1. `mark_session_deleted` 的 UPDATE 语句加防降级条件：

```python
    cur = conn.execute(
        "UPDATE sessions SET status='deleted', updated_at=? "
        "WHERE session_id=? AND status != 'purged'",
        (now, session_id),
    )
```

2. `upsert_topic` 开头守卫——把原有第一行查询改为同时取 status，并紧接守卫（其余逻辑不动）：

```python
    existing = conn.execute(
        "SELECT updated_at, status FROM sessions WHERE session_id=?",
        (rec.session_id,),
    ).fetchone()
    if existing is not None and existing["status"] == "purged":
        # 守卫①：purged 终态不可复活（import/rebuild 旧数据也走这条路）
        return WriteResult(False, False)
```

3. `upsert_message` 开头加独立守卫（**必须在 `INSERT OR IGNORE INTO messages` 之前**，此时还没有 `existing` 查询可复用）：

```python
    guarded = conn.execute(
        "SELECT status FROM sessions WHERE session_id=?", (rec.session_id,)
    ).fetchone()
    if guarded is not None and guarded["status"] == "purged":
        return WriteResult(False, False)
```

```python
                    catalog.insert_session_purge_marker(
                        conn, session_id=session_id,
                        project_id=meta["project_id"],
                        source_kind=meta.get("source_kind") or "session",
                        created_at=meta.get("created_at") or "",
                        updated_at=meta.get("updated_at")
                        or meta.get("created_at") or "",
                    )
                    sessions += 1
                    continue
```

```python
"""purge 后对既有归档包做原位剔除（Spec 13 D5）。

纯文件操作：不碰数据库、零 SQL 字面量。先 verify，再逐条复制原包条目——只把
目标会话的 meta 换成 purged 精简卡、ndjson 换成 0 字节空文件（verify 的
meta↔ndjson 配对不变量不允许只删一条），其余条目字节级原样——manifest 重算
files/counts/archive_id 后临时落地、自检、原子改名。旧包删除由调用方负责
（本函数失败时旧包原样在场，可回退）。
"""
from __future__ import annotations

import json
import os
import zipfile
from pathlib import Path

# 同包互导私有工具函数属 lint_imports 豁免（E_PRIVATE 同包豁免）
from sessionpack.archive.build import (MANIFEST_NAME, _sha256_bytes, _slug,
                                       compute_archive_id)
from sessionpack.archive.verify import ArchiveError, verify_archive


class RepakError(Exception):
    """重打包前置条件不满足（包内没有目标会话等）。"""


def _purged_card(old_meta: dict, session_id: str) -> dict:
    """purged 精简卡：只留身份与时间字段，不含 title/summary 等内容字段。"""
    return {
        "session_id": session_id,
        "project_id": old_meta["project_id"],
        "source_kind": old_meta.get("source_kind") or "session",
        "status": "purged",
        "created_at": old_meta.get("created_at") or "",
        "updated_at": (old_meta.get("updated_at")
                       or old_meta.get("created_at") or ""),
    }


def repak_excluding_session(src: Path, session_id: str, dest_dir: Path) -> Path:
    """把 src 包中 session_id 替换为 purge 卡后写入 dest_dir，返回新包路径。"""
    check = verify_archive(src)
    if not check.ok:
        raise ArchiveError(check.issues)
    card_name = f"sessions/{session_id}/meta.json"
    ndjson_name = f"sessions/{session_id}/messages.ndjson"
    entries: list[tuple[str, bytes]] = []
    with zipfile.ZipFile(src) as zf:
        manifest = json.loads(zf.read(MANIFEST_NAME))
        if card_name not in zf.namelist():
            raise RepakError(f"包内无会话 {session_id}")
        card = _purged_card(json.loads(zf.read(card_name)), session_id)
        card_bytes = (json.dumps(card, ensure_ascii=False, indent=2)
                      + "\n").encode("utf-8")
        for info in zf.infolist():
            if info.filename == MANIFEST_NAME:
                continue                      # manifest 统一重建
            if info.filename == card_name:
                entries.append((card_name, card_bytes))
            elif info.filename == ndjson_name:
                entries.append((ndjson_name, b""))   # 空 ndjson 合法
            else:
                entries.append((info.filename, zf.read(info.filename)))
    files = [{"path": p, "sha256": _sha256_bytes(b), "size": len(b)}
             for p, b in entries]
    sessions = sum(1 for p, _ in entries
                   if p.startswith("sessions/") and p.endswith("/meta.json"))
    messages = sum(1 for p, b in entries if p.endswith("messages.ndjson")
                   for line in b.splitlines() if line.strip())
    archive_id = compute_archive_id(files)
    manifest["archive_id"] = archive_id
    manifest["files"] = files
    manifest["counts"] = {"sessions": sessions, "messages": messages}
    manifest_bytes = (json.dumps(manifest, ensure_ascii=False, indent=2)
                      + "\n").encode("utf-8")
    dest_dir.mkdir(parents=True, exist_ok=True)
    temp = dest_dir / f".tmp-repak-{archive_id[:12]}.zip"
    with zipfile.ZipFile(temp, "w", zipfile.ZIP_DEFLATED) as zf:
        for name, data in entries:
            zf.writestr(name, data)
        zf.writestr(MANIFEST_NAME, manifest_bytes)
    recheck = verify_archive(temp)
    if not recheck.ok:
        temp.unlink(missing_ok=True)
        raise ArchiveError(recheck.issues)
    # 日期段沿用原包 created_at：本机改名只由 archive_id 段体现
    date_seg = (manifest.get("created_at") or "")[:10]
    final = dest_dir / (f"{_slug(manifest['project']['project_id'])}"
                        f"-{date_seg}-{archive_id[:12]}.zip")
    os.replace(temp, final)
    return final
```

```python
def find_archives_with_session(directory: Path, session_id: str) -> list[str]:
    """只读探测：directory 下含该会话**内容**的包文件名（Spec 13 D6 报告用）。

    已替换为 purge 卡的包（meta status='purged'、正文为空）不算命中——本机
    重打包循环与同步目录报告都只关心"还带着内容"的包。读不了的包按"不含"
    跳过：报告场景宁漏勿错，不阻塞命令。
    """
    if not directory.is_dir():
        return []
    card = f"sessions/{session_id}/meta.json"
    hits: list[str] = []
    for path in _scan_zips(directory):
        try:
            with zipfile.ZipFile(path) as zf:
                if card not in zf.namelist():
                    continue
                if json.loads(zf.read(card)).get("status") == "purged":
                    continue
                hits.append(path.name)
        except (OSError, zipfile.BadZipFile, KeyError, ValueError):
            continue
    return hits
```

```python
def _run_purge(conn, lay, args, status: str) -> CommandResult:
    """delete --purge --yes 的编排（Spec 13 D4）：先库后包、单包失败不回滚。

    调用方 run_delete 已完成索引存在性、会话存在性（status 非 None）与连接管理。
    """
    if getattr(args, "undo", False):
        return CommandResult(EXIT_USAGE,
                             err_lines=["--purge 与 --undo 不能同时使用"])
    if not getattr(args, "yes", False):
        return CommandResult(EXIT_USAGE, err_lines=[
            "--purge 会物理抹除本机内容且不可逆，请加 --yes 确认"])
    if status == "active":
        return CommandResult(EXIT_USAGE, err_lines=[
            f"需先软删除再清除: sessionpack delete --session {args.session}"])
    # status 为 deleted 或 purged（G2：只从 deleted 进；purged 走幂等成功）
    messages_removed = (len(catalog.get_messages(conn, args.session))
                        if status == "deleted" else 0)
    if status == "deleted":
        catalog.purge_session(
            conn, args.session,
            now=datetime.now().strftime("%Y-%m-%dT%H:%M:%S"))
        conn.commit()
    repacked: list[str] = []
    failed: list[str] = []
    for name in sync.find_archives_with_session(lay.archives, args.session):
        try:
            new = archive.repak_excluding_session(
                lay.archives / name, args.session, lay.archives)
        except (archive.ArchiveError, archive.RepakError, OSError) as exc:
            failed.append(f"{name}: {exc}")
            continue
        (lay.archives / name).unlink()
        repacked.append(new.name)
    sync_dir = resolve_sync_dir(args, lay)
    sync_stale = (sync.find_archives_with_session(sync_dir, args.session)
                  if sync_dir else [])
    data = {"session_id": args.session, "purged": True,
            "messages_removed": messages_removed,
            "archives_repacked": len(repacked),
            "archives_failed": failed, "sync_stale": sync_stale}
    lines = [f"已清除会话 {args.session}：抹除消息 {messages_removed} 条，"
             f"重打包 {len(repacked)} 个"]
    lines += [f"重打包失败（旧包原样在场）: {item}" for item in failed]
    lines += [f"同步目录仍含旧包（本命令不改动，请手动处理）: {item}"
              for item in sync_stale]
    if failed:
        return CommandResult(EXIT_ERROR, out_lines=lines, data=data)
    return CommandResult(EXIT_OK, out_lines=lines, data=data)
```

```python
def test_rebuild_keeps_purge_card(tmp_path):
    """purge → rebuild --from-archives → 终态卡仍在、正文不回。"""
    root, lay = _purge_seeded(tmp_path)
    assert _purge_cli(root, SID).exit_code == 0
    report = rebuild_from_archives(lay, assume_yes=True)
    assert report.ok, report.error
    conn = catalog.connect(lay.catalog)
    assert catalog.get_session_status(conn, SID) == "purged"
    assert catalog.get_session(conn, SID) is None
    assert catalog.get_messages(conn, SID) == []
    conn.close()


def test_cli_purge_reports_stale_sync_dir(tmp_path):
    """同步目录含旧包 → 只报告不改动，命令仍成功。"""
    root, lay = _purge_seeded(tmp_path)
    stale = tmp_path / "sync"
    stale.mkdir()
    names = sorted(p.name for p in lay.archives.glob("*.zip"))
    for item in lay.archives.glob("*.zip"):
        (stale / item.name).write_bytes(item.read_bytes())
    res = _purge_cli(root, SID, json_out=True, directory=str(stale))
    assert res.exit_code == 0
    assert res.data["sync_stale"] == names
    assert res.data["archives_repacked"] == 1
    # 同步目录内容原样未动
    assert sorted(p.name for p in stale.glob("*.zip")) == names
```
