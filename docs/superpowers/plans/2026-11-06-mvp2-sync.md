# MVP 2 用户自选目录同步端 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 SessionPack 增加 `push` / `pull` 两条 CLI 命令，实现本地 `archives/` 与用户自选同步目录之间的单向只增同步（pull = 复制 + 自动导入），并落档 MIT 许可证。

**Architecture:** 新增 `sessionpack/sync.py`（纯文件操作 + 既有 `verify_archive` / `import_archive` 编排，不碰 SQLite schema）与 `sessionpack/config.py`（stdlib `tomllib` 读 + 手写序列化写 `[sync] dir`）。CLI 注册两个子命令，目录指针持久化到 `<data-root>/config.toml`。`diagnostics.collect()` 增加 `sync` 段；诊断页现状为整包 JSON 渲染，sync 段自动呈现，前端零改动。不改 `archive.py` / `importer.py` / `catalog.py` 既有逻辑，`paths.Layout` 不加字段。

**Tech Stack:** Python 3.11+ 标准库（`tomllib` / `shutil` / `os.replace` / `argparse`）、pytest 8、既有 SQLite 目录模块。

## Global Constraints

- 运行时零第三方依赖：`pyproject.toml` 的 `dependencies = []` 不变，不引入 `tomli-w`。
- Python `>=3.11`（`tomllib` 为 stdlib）。
- 退出码沿用 `sessionpack/cli.py` 既有常量：`EXIT_OK=0`、`EXIT_ERROR=1`、`EXIT_USAGE=2`、`EXIT_VERIFY=3`。
- 同步产生的错误文本一律经 `sessionpack.archive._failure_text` 净化（绝对路径→`<路径>`，限长 60），不得泄漏绝对路径。
- CLI 面向用户的输出为中文；错误提示走 `sys.stderr`。
- 同步目录平铺 `<slug>-<date>-<id12>.zip`，不加子目录、不改名；`push` / `pull` 只读写 `*.zip` 与 `.tmp-<name>` 临时文件，不遍历、不删除同步目录内其他文件。
- 两端只增不删：任何同步命令不删除任何一端的既有文件。
- 全量回归：`python -m pytest -q`；当前基线 **110 passed**（2026-11-06 实测）。
- 每个任务完成后直接在 `main` 提交（用户已批准直接提交 main 并推送）。

```python
from __future__ import annotations

import json
import tomllib
from pathlib import Path

CONFIG_NAME = "config.toml"


def load_config(root: Path) -> tuple[dict, list[str]]:
    """读取 <root>/config.toml；缺失或损坏按空配置处理并返回警告。"""
    path = Path(root) / CONFIG_NAME
    if not path.is_file():
        return {}, []
    try:
        with path.open("rb") as handle:
            data = tomllib.load(handle)
    except (tomllib.TOMLDecodeError, OSError):
        return {}, ["config.toml 解析失败，已按空配置处理"]
    return (data, []) if isinstance(data, dict) else ({}, [])


def sync_dir(root: Path) -> Path | None:
    data, _ = load_config(root)
    section = data.get("sync")
    if not isinstance(section, dict):
        return None
    raw = section.get("dir")
    if not isinstance(raw, str) or not raw:
        return None
    return Path(raw).expanduser()


def set_sync_dir(root: Path, directory: Path) -> None:
    root = Path(root)
    root.mkdir(parents=True, exist_ok=True)
    text = "[sync]\ndir = " + json.dumps(str(Path(directory))) + "\n"
    (root / CONFIG_NAME).write_text(text, encoding="utf-8")
```

创建 `sessionpack/sync.py`：

注意 import 风格：统一用 `from . import archive` 模块引用（而非 `from .archive import verify_archive`），这样 Task 3 测试中 `monkeypatch.setattr(archive, "import_archive", ...)` 才能命中 `sync` 模块实际调用的函数。

```python
from __future__ import annotations

import os
import re
import shutil
from dataclasses import dataclass
from pathlib import Path

from . import archive

_TAIL_RE = re.compile(r"-([0-9a-f]{12})\.zip$")


@dataclass
class SyncResult:
    pushed: int = 0
    pulled: int = 0
    skipped_existing: int = 0
    skipped_unverified: tuple[str, ...] = ()
    skipped_invalid: tuple[str, ...] = ()
    sessions_added: int = 0
    messages_added: int = 0
    bytes: int = 0
    errors: tuple[str, ...] = ()


def _id_tail(name: str) -> str | None:
    match = _TAIL_RE.search(name)
    return match.group(1) if match else None


def _atomic_copy(src: Path, dest: Path) -> None:
    dest.parent.mkdir(parents=True, exist_ok=True)
    tmp = dest.parent / (".tmp-" + dest.name)
    try:
        shutil.copy2(src, tmp)
        os.replace(tmp, dest)
    finally:
        tmp.unlink(missing_ok=True)


def push_archives(archives_dir: Path, sync_dir: Path,
                  dry_run: bool = False) -> SyncResult:
    pushed: list[str] = []
    existing = 0
    unverified: list[str] = []
    errors: list[str] = []
    total = 0
    local = sorted(Path(archives_dir).glob("*.zip"))
    remote_names = (
        {p.name for p in Path(sync_dir).glob("*.zip")}
        if Path(sync_dir).is_dir() else set()
    )
    for src in local:
        if src.name in remote_names:
            existing += 1
            continue
        check = archive.verify_archive(src)
        if not check.ok:
            codes = ", ".join(issue.code for issue in check.issues)
            unverified.append(f"{src.name}: {codes}")
            continue
        if not dry_run:
            try:
                _atomic_copy(src, Path(sync_dir) / src.name)
            except OSError as exc:
                errors.append(f"{src.name}: {archive._failure_text(exc)}")
                continue
        pushed.append(src.name)
        total += src.stat().st_size
    return SyncResult(pushed=len(pushed), skipped_existing=existing,
                      skipped_unverified=tuple(unverified),
                      bytes=total, errors=tuple(errors))
```

说明：`dry_run=True` 时不调 `_atomic_copy`（也就不会 mkdir），只按 verify 结果统计"将复制"清单——与规格"只打印将复制的文件清单与字节数，不写任何东西"一致。

修改 `sessionpack/sync.py`（Task 2 已用 `from . import archive` 模块引用风格，`import_archive` 直接经 `archive.import_archive` 调用，测试的 monkeypatch 目标即 `archive.import_archive`，无需再改 import）。在文件末尾追加：

```python
def pull_archives(archives_dir: Path, sync_dir: Path, conn) -> SyncResult:
    archives_dir = Path(archives_dir)
    pulled = 0
    existing = 0
    invalid: list[str] = []
    errors: list[str] = []
    sessions_added = 0
    messages_added = 0
    total = 0
    local_ids = {tail for p in archives_dir.glob("*.zip")
                 if (tail := _id_tail(p.name))}
    for src in sorted(Path(sync_dir).glob("*.zip")):
        tail = _id_tail(src.name)
        if tail and tail in local_ids:
            existing += 1
            continue
        check = archive.verify_archive(src)
        if not check.ok:
            codes = ", ".join(issue.code for issue in check.issues)
            invalid.append(f"{src.name}: {codes}")
            continue
        size = src.stat().st_size
        dest = archives_dir / src.name
        try:
            _atomic_copy(src, dest)
        except OSError as exc:
            errors.append(f"{src.name}: {archive._failure_text(exc)}")
            continue
        try:
            imported = archive.import_archive(conn, dest)
        except (archive.ArchiveError, OSError) as exc:
            errors.append(f"{src.name}: 已复制但导入失败 "
                          f"{archive._failure_text(exc)}")
            continue
        pulled += 1
        total += size
        sessions_added += imported.sessions_created
        messages_added += imported.messages_inserted
        if tail:
            local_ids.add(tail)
    return SyncResult(pulled=pulled, skipped_existing=existing,
                      skipped_invalid=tuple(invalid),
                      sessions_added=sessions_added,
                      messages_added=messages_added,
                      bytes=total, errors=tuple(errors))
```

```python
def _resolve_sync_dir(args, layout: Layout) -> Path | None:
    if args.dir:
        return Path(args.dir).expanduser()
    return config.sync_dir(layout.root)


def _cmd_push(args) -> int:
    layout = _layout(args)
    target = _resolve_sync_dir(args, layout)
    if target is None:
        print("未指定同步目录：请加 --dir DIR，或先运行 sessionpack push --dir DIR 落配置",
              file=sys.stderr)
        return EXIT_USAGE
    result = sync.push_archives(layout.archives, target, dry_run=args.dry_run)
    if args.json:
        print(json.dumps({
            "sync_dir": str(target),
            "pushed": result.pushed,
            "skipped_existing": result.skipped_existing,
            "skipped_unverified": list(result.skipped_unverified),
            "bytes": result.bytes,
            "errors": list(result.errors),
        }, ensure_ascii=False, indent=2))
    else:
        verb = "将推送" if args.dry_run else "已推送"
        print(f"[dry-run] " if args.dry_run else ""
              + f"{verb} {result.pushed} 个包共 {result.bytes} 字节，"
                f"跳过已存在 {result.skipped_existing}，"
                f"跳过未通过校验 {len(result.skipped_unverified)}")
        for item in result.skipped_unverified:
            print(f"跳过未通过校验的包: {item}")
        for item in result.errors:
            print(f"推送失败: {item}", file=sys.stderr)
    if result.errors:
        return EXIT_ERROR
    if args.dir and not args.dry_run:
        config.set_sync_dir(layout.root, target)
    return EXIT_OK


def _cmd_pull(args) -> int:
    layout = _layout(args)
    target = _resolve_sync_dir(args, layout)
    if target is None:
        print("未指定同步目录：请加 --dir DIR，或先运行 sessionpack push --dir DIR 落配置",
              file=sys.stderr)
        return EXIT_USAGE
    if not target.is_dir():
        print(f"同步目录不存在: {target}", file=sys.stderr)
        return EXIT_USAGE
    conn = catalog.connect(layout.catalog)
    catalog.init_schema(conn)
    try:
        result = sync.pull_archives(layout.archives, target, conn)
    finally:
        conn.close()
    if args.json:
        print(json.dumps({
            "sync_dir": str(target),
            "pulled": result.pulled,
            "skipped_existing": result.skipped_existing,
            "skipped_invalid": list(result.skipped_invalid),
            "sessions_added": result.sessions_added,
            "messages_added": result.messages_added,
            "bytes": result.bytes,
            "errors": list(result.errors),
        }, ensure_ascii=False, indent=2))
    else:
        print(f"已拉取 {result.pulled} 个包（新增会话 {result.sessions_added}、"
              f"消息 {result.messages_added}），跳过已存在 {result.skipped_existing}，"
              f"跳过无效包 {len(result.skipped_invalid)}")
        for item in result.skipped_invalid:
            print(f"跳过无效包: {item}")
        for item in result.errors:
            print(f"拉取失败: {item}", file=sys.stderr)
    return EXIT_ERROR if result.errors else EXIT_OK
```

实现注意：
- `catalog.init_schema(conn)` 是幂等的（全 `CREATE ... IF NOT EXISTS`，见 `sessionpack/catalog.py`），fresh 与已有索引都可安全调用。
- 上面 push 非 json 分支的 `print` 写法为便于阅读拆开，落地时写成一个 f-string；`[dry-run] ` 前缀仅 dry_run 时加。
- `_cmd_pull` 的 fresh 场景（data-dir 无 catalog.sqlite）依赖 `catalog.connect` 建文件 + `init_schema` 建表；`_catalog_ready` 预检**不适用**于 pull（有意区别于 export/serve）。

def _sync_section(root: Path) -> dict:
    target = config.sync_dir(root)
    if target is None:
        return {"configured": False, "dir": None, "exists": False,
                "archives": 0, "bytes": 0}
    exists = target.is_dir()
    files = [p for p in target.glob("*.zip") if p.is_file()] if exists else []
    return {"configured": True, "dir": str(target), "exists": exists,
            "archives": len(files),
            "bytes": sum(p.stat().st_size for p in files)}
