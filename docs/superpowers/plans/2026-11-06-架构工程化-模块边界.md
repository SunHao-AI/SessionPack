> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用一条 import/SQL 守卫（`scripts/lint_imports.py` + 只减不增基线）机械锁定模块边界，并按 P1→P5→P2→P4→P3 六个 commit 修复全部已知边界违规，基线清零。

**Architecture:** 先落地守卫（C1 时基线收录当时全部 21 条违规保证全绿；spec 表列 22 条，其中 archive build 段 3 处为纯 SELECT 不被写语句正则命中，实际基线 21 条），此后每个修复 commit 同步删除对应基线条目作为完成证明；importer 的裸 SQL 收敛到 catalog 数据访问 API，`_failure_text` 公开为 `textutil.failure_text`，archive.py 拆为 `archive/` 包且对外 import 路径与符号不变。

**Tech Stack:** Python 3.10+，纯 stdlib（re/sqlite3/zipfile/argparse），无新第三方依赖；质量链 ruff + mypy + pytest。

## Global Constraints

- 规格唯一依据：`docs/superpowers/specs/2026-11-06-架构工程化-模块边界-design.md`。
- 零第三方新增依赖；守卫脚本仅用 stdlib。

- 144 个现有测试不删、不改断言；新增测试后应为 147 passed。

```python
"""模块边界守卫：层级白名单 / 私有符号跨模块 / SQL 独占。仅 stdlib。

用法:  python scripts/lint_imports.py [--update]
--update 会把当前违规写回本文件的 BASELINE（只允许在违规确属预期时手动使用）。
输出格式: 文件:行: 规则名 说明
"""
from __future__ import annotations

import ast
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
PKG_DIR = ROOT / "sessionpack"
BASELINE_FILE = Path(__file__)

# §3.2 分层规则表：数字越大层级越高；同层允许。
LAYER_OF: dict[str, int] = {
    "model": 0, "paths": 0, "audit": 0, "mem": 0, "summary": 0,
    "config": 0, "textutil": 0,
    "adapters": 1,
    "catalog": 2,
    "archive": 3, "export": 3, "importer": 3, "sync": 3, "diagnostics": 3,
    "cli": 4, "server": 4,
}

SQL_WRITE = re.compile(
    r"(INSERT|UPDATE|DELETE FROM|CREATE TABLE).*(projects|sessions|messages"
    r"|import_runs|import_files|message_fts)"
)

# 只减不增：修复一个删一行；禁止向其中新增条目来"解决"新违规。
BASELINE: set[str] = {
    "sessionpack/importer.py:12: E_LAYER import L3 -> L1 (importer -> adapters.trae_memory)",
    "sessionpack/importer.py:128: E_PRIVATE module attribute 'project_display_name' imported from 'sessionpack.adapters.trae_memory' is private",
    "sessionpack/diagnostics.py:8: E_LAYER import L3 -> L1 (diagnostics -> adapters.trae_memory)",
    "sessionpack/server.py:144: E_PRIVATE module attribute '_render' called on 'sessionpack.export'",
    "sessionpack/sync.py:74: E_PRIVATE module attribute '_failure_text' called on 'sessionpack.archive'",
    "sessionpack/sync.py:80: E_PRIVATE module attribute '_failure_text' called on 'sessionpack.archive'",
    "sessionpack/sync.py:118: E_PRIVATE module attribute '_failure_text' called on 'sessionpack.archive'",
    "sessionpack/sync.py:124: E_PRIVATE module attribute '_failure_text' called on 'sessionpack.archive'",
    "sessionpack/archive.py:126: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/archive.py:131: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/archive.py:143: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/archive.py:363: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/archive.py:416: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/archive.py:425: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/importer.py:112: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/importer.py:173: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/importer.py:184: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/importer.py:189: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/importer.py:205: E_SQL raw SQL write on managed table (only catalog.py allowed)",
    "sessionpack/importer.py:294: E_SQL raw SQL write on managed table (only catalog.py allowed)",
}


def module_of(rel: str) -> tuple[str | None, str | None]:
    """'sessionpack/importer.py' -> ('importer', None)；
    'sessionpack/adapters/trae_memory.py' -> ('adapters', 'trae_memory')；
    'sessionpack/__init__.py' -> (None, None)。"""
    parts = rel.removesuffix(".py").split("/")
    if parts[0] != "sessionpack" or len(parts) < 2:
        return None, None
    head = parts[1]
    if head == "__init__":
        return None, None
    if len(parts) == 2:
        return head, None
    if head == "adapters":
        return "adapters", parts[2]
    return head, parts[-1]  # 未来其他子包：首段为包名


def layer(rel_or_mod: str) -> int | None:
    mod, sub = module_of(rel_or_mod)
    if mod is None:
        return None
    return LAYER_OF.get(sub or mod, LAYER_OF.get(mod))


def _abs_from_import(node, source_module: str) -> str:
    """把 ImportFrom 解析为绝对模块点路径（相对导入按源模块位置解析）。"""
    if node.level == 0:
        return node.module or ""
    base = "sessionpack"
    if source_module == "__init__":
        target = base
    else:
        target = f"{base}.{source_module}"
    for _ in range(node.level - 1):
        target = target.rpartition(".")[0] or base
    return f"{target}.{node.module}" if node.module else target


def check_source(src: str, rel: str) -> list[str]:
    """对单个文件源码返回违规列表（字符串格式见 BASELINE）。"""
    tree = ast.parse(src)
    mod, _ = module_of(rel)
    source_module = mod or "__init__"
    out: list[str] = sql_violations(tree, rel)

    imported_names: dict[str, str] = {}  # 本地名 -> 定义它的绝对模块
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for a in node.names:
                if a.name.startswith("sessionpack"):
                    imported_names[a.asname or a.name.split(".")[0]] = a.name
        elif isinstance(node, ast.ImportFrom):
            amod = _abs_from_import(node, source_module)
            for a in node.names:
                local = a.asname or a.name
                if a.name.startswith("_"):
                    owner = amod.rpartition(".")[0]
                    if (amod.startswith("sessionpack")
                            and not owner.endswith(source_module)
                            and owner != "sessionpack"):
                        out.append(
                            f"{rel}:{node.lineno}: E_PRIVATE 'from import' "
                            f"pulls private '{a.name}' from '{amod}'")
                if amod.startswith("sessionpack"):
                    if "." in amod:
                        imported_names[local] = amod
                    else:
                        imported_names[local] = amod
    for local, owner in list(imported_names.items()):
        if local.startswith("_") and not owner.endswith(source_module):
            out.append(
                f"{rel}:0: E_PRIVATE module attribute '{local}' imported "
                f"from '{owner}' is private")

    # 属性访问: obj._attr(...) / obj._attr
    for node in ast.walk(tree):
        if isinstance(node, ast.Attribute) and node.attr.startswith("_"):
            v = node.value
            if isinstance(v, ast.Name) and v.id in imported_names:
                owner = imported_names[v.id]
                owner_tail = owner.removeprefix("sessionpack.").split(".")[0]
                if owner_tail != source_module and owner_tail != "sessionpack":
                    out.append(
                        f"{rel}:{node.lineno}: E_PRIVATE module attribute "
                        f"'{node.attr}' called on '{owner}'")

    # 层级白名单
    my_layer = layer(rel)
    for node in ast.walk(tree):
        mods: list[str] = []
        if isinstance(node, ast.Import):
            mods = [a.name for a in node.names]
        elif isinstance(node, ast.ImportFrom):
            amod = _abs_from_import(node, source_module)
            mods = [amod] + [f"{amod}.{a.name}" for a in node.names]
        for m in mods:
            if not m.startswith("sessionpack"):
                continue
            parts = m.split(".")
            if len(parts) < 2 or parts[1] == "__init__":
                continue
            tgt_top = parts[1]
            tgt = parts[2] if (tgt_top == "adapters" and len(parts) > 2) else tgt_top
            tgt_layer = LAYER_OF.get(tgt, LAYER_OF.get(tgt_top))
            if my_layer is None or tgt_layer is None:
                continue
            if parts[1:2] and source_module == tgt_top:
                continue  # 包内/同模块边豁免
            if tgt_layer > my_layer:
                out.append(
                    f"{rel}:{node.lineno}: E_LAYER import "
                    f"L{my_layer} -> L{tgt_layer} ({source_module} -> {tgt})")
    return sorted(set(out), key=lambda s: (s.split(":")[0], int(s.split(":")[1])))


def sql_violations(tree: ast.AST, rel: str) -> list[str]:
    """SQL 独占令：字符串字面量含受管表写语句，仅 catalog.py 允许。"""
    if rel == "sessionpack/catalog.py":
        return []
    out: list[str] = []
    for node in ast.walk(tree):
        if (isinstance(node, ast.Constant) and isinstance(node.value, str)
                and SQL_WRITE.search(node.value)):
            out.append(f"{rel}:{node.lineno}: E_SQL raw SQL write on "
                       "managed table (only catalog.py allowed)")
    # SQL 常量可能被变量引用后传入 execute；AST 字面量检索对多段拼接
    # 字符串已在 f-string/常量折叠下可检，若发现漏网请改用 tokenize 全文扫描。
    return out


def check_tree(root: Path) -> list[str]:
    found: list[str] = []
    for path in sorted((root / "sessionpack").rglob("*.py")):
        rel = path.relative_to(root).as_posix()
        found.extend(check_source(path.read_text(encoding="utf-8"), rel))
    return found


def _update_baseline(violations: set[str]) -> None:
    text = BASELINE_FILE.read_text(encoding="utf-8")
    lines = "".join(f'    "{v}",\n' for v in sorted(violations))
    new = re.sub(r"BASELINE: set\[str\] = \{\n.*?\n\}\n",
                 f"BASELINE: set[str] = {{\n{lines}}}\n", text, count=1,
                 flags=re.S)
    BASELINE_FILE.write_text(new, encoding="utf-8")


def main(argv: list[str]) -> int:
    found = check_tree(ROOT)
    found_set = set(found)
    if "--update" in argv:
        _update_baseline(found_set)
        print(f"BASELINE updated: {len(found_set)} entries")
        return 0
    for v in found:
        print(v)
    new = found_set - BASELINE
    gone = BASELINE - found_set
    if gone:
        print(f"!! {len(gone)} 条基线违规已消除，请删除 BASELINE 对应条目"
              "（这是好事，说明修复生效）", file=sys.stderr)
    if new:
        print(f"!! {len(new)} 条新增违规（基线只减不增，禁止 --update 绕过）",
              file=sys.stderr)
        return 1
    if gone:
        return 1
    print(f"lint_imports OK ({len(found_set)} 条基线违规，无新增)")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
# ---------------------------------------------------------------- run / file 表


def begin_import_run(conn: sqlite3.Connection, run_id: str, adapter_id: str,
                     adapter_version: str, started_at: str) -> None:
    conn.execute(
        "INSERT INTO import_runs(run_id, adapter_id, adapter_version,"
        " started_at, result) VALUES(?,?,?,?,?)",
        (run_id, adapter_id, adapter_version, started_at, "running"))


def finish_import_run(conn: sqlite3.Connection, run_id: str, *,
                      finished_at: str, result: str,
                      scanned_files: int | None = None,
                      imported_rows: int | None = None,
                      failed_rows: int | None = None,
                      warnings_json: str | None = None) -> None:
    cols: dict[str, object] = {
        "finished_at": finished_at, "result": result,
        "scanned_files": scanned_files, "imported_rows": imported_rows,
        "failed_rows": failed_rows, "warnings_json": warnings_json,
    }
    parts = ", ".join(f"{k}=?" for k, v in cols.items() if v is not None)
    args = [v for v in cols.values() if v is not None]
    conn.execute(f"UPDATE import_runs SET {parts} WHERE run_id=?",
                 (*args, run_id))


def load_import_files(conn: sqlite3.Connection) -> dict[str, sqlite3.Row]:
    return {row["path"]: row
            for row in conn.execute("SELECT * FROM import_files")}


def register_import_files(conn: sqlite3.Connection,
                          rows: list[tuple[str, int, float]]) -> None:
    conn.executemany(
        "INSERT OR IGNORE INTO import_files(path, size, mtime) VALUES(?,?,?)",
        rows)


def reset_import_file(conn: sqlite3.Connection, *, size: int, mtime: float,
                      path: str) -> None:
    conn.execute(
        "UPDATE import_files SET size=?, mtime=?, offset=0, rows=0,"
        " done=0, error=NULL WHERE path=?", (size, mtime, path))


def clear_import_file_error(conn: sqlite3.Connection, path: str) -> None:
    conn.execute(
        "UPDATE import_files SET offset=0, rows=0, done=0, error=NULL"
        " WHERE path=?", (path,))


def save_import_file_state(conn: sqlite3.Connection, *, path: str, offset: int,
                           rows: int, done: int, last_run_id: str,
                           error: str | None) -> None:
    conn.execute(
        "UPDATE import_files SET offset=?, rows=?, done=?, last_run_id=?,"
        " error=? WHERE path=?",
        (offset, rows, done, last_run_id, error, path))
```

```python
"""失败信息摘要：OSError 友好化 + 绝对路径脱敏 + 截断。L0 叶子模块。"""
from __future__ import annotations

import re

_PATH_TEXT_RE = re.compile(
    r"[A-Za-z]:[\\/][^\s\"']*|/(?:Users|home|var|tmp)[^\s\"']*")


def failure_text(exc: BaseException, limit: int = 60) -> str:
    if isinstance(exc, OSError) and exc.strerror:
        text = exc.strerror
        if exc.filename:
            text += "（文件位于数据目录内）"
    else:
        text = str(exc)
    return _PATH_TEXT_RE.sub("<路径>", text)[:limit]
```
