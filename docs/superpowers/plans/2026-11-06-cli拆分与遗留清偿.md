```python
"""catalog 的连接、建表与元信息。

可写连接是 connect()，只读连接是 connect_readonly()：
后者把散落在 cli/diagnostics 的 mode=ro 样板收口到一处。
"""

from __future__ import annotations

import re
import sqlite3
from pathlib import Path

SESSION_ID_RE = re.compile(r"[0-9a-f]{24}")

SCHEMA = """
CREATE TABLE IF NOT EXISTS projects (
  project_id      TEXT PRIMARY KEY,
  display_name    TEXT NOT NULL,
  local_path_hint TEXT,
  source_kind     TEXT NOT NULL,
  updated_at      TEXT
);
CREATE TABLE IF NOT EXISTS sessions (
  session_id  TEXT PRIMARY KEY,
  project_id  TEXT NOT NULL,
  source_kind TEXT NOT NULL,
  title       TEXT,
  summary     TEXT,
  created_at  TEXT,
  updated_at  TEXT,
  content_hash TEXT,
  status      TEXT NOT NULL DEFAULT 'active'
);
CREATE TABLE IF NOT EXISTS messages (
  session_id TEXT NOT NULL,
  message_id TEXT NOT NULL,
  seq        INTEGER NOT NULL,
  ts         TEXT,
  intent     TEXT,
  outcome    TEXT,
  actions    TEXT,
  learned    TEXT,
  meta_json  TEXT,
  body       TEXT,
  PRIMARY KEY (session_id, message_id)
);
CREATE TABLE IF NOT EXISTS import_runs (
  run_id         TEXT PRIMARY KEY,
  adapter_id     TEXT NOT NULL,
  adapter_version INTEGER NOT NULL,
  started_at     TEXT NOT NULL,
  finished_at    TEXT,
  result         TEXT NOT NULL,
  scanned_files  INTEGER NOT NULL DEFAULT 0,
  imported_rows  INTEGER NOT NULL DEFAULT 0,
  failed_rows    INTEGER NOT NULL DEFAULT 0,
  warnings_json  TEXT
);
CREATE TABLE IF NOT EXISTS import_files (
  path        TEXT PRIMARY KEY,
  size        INTEGER NOT NULL,
  mtime       REAL NOT NULL,
  offset      INTEGER NOT NULL DEFAULT 0,
  rows        INTEGER NOT NULL DEFAULT 0,
  done        INTEGER NOT NULL DEFAULT 0,
  last_run_id TEXT,
  error       TEXT
);
CREATE INDEX IF NOT EXISTS idx_sessions_project ON sessions(project_id, updated_at);
CREATE INDEX IF NOT EXISTS idx_messages_session ON messages(session_id, seq);
"""


def connect(path: Path) -> sqlite3.Connection:
    path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(path, timeout=30)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA synchronous=NORMAL")
    return conn


def connect_readonly(path: Path) -> sqlite3.Connection:
    """只读连接（URI mode=ro）：不建目录、不建库。

    文件缺失/非 sqlite 文件/权限不足都抛 sqlite3.Error，由调用方捕获。
    """
    conn = sqlite3.connect(f"file:{path.resolve().as_posix()}?mode=ro", uri=True)
    conn.row_factory = sqlite3.Row
    return conn


def init_schema(conn: sqlite3.Connection) -> str:
    conn.executescript(SCHEMA)
    conn.execute(
        "CREATE VIRTUAL TABLE IF NOT EXISTS message_fts USING fts5("
        "session_id UNINDEXED, message_id UNINDEXED, text, tokenize='trigram')"
    )
    conn.commit()
    return tokenizer_of(conn)


def tokenizer_of(conn: sqlite3.Connection) -> str:
    row = conn.execute(
        "SELECT sql FROM sqlite_master WHERE name='message_fts'"
    ).fetchone()
    if row is None:
        return "none"
    return "trigram" if "trigram" in (row[0] or "") else "unicode61"
```

```python
"""catalog 的只读查询：检索、列表、详情、统计与只读表加载。

本模块只做 SELECT；SQL 守卫会拦住任何误写的 INSERT/UPDATE/DELETE。
"""

from __future__ import annotations

import sqlite3


def has_schema(conn: sqlite3.Connection) -> bool:
    row = conn.execute(
        "SELECT count(*) FROM sqlite_master WHERE type='table' AND name='projects'"
    ).fetchone()
    return bool(row and row[0])


def _fts_expr(term: str) -> str:
    return '"' + term.replace('"', '""') + '"'


def _like(term: str) -> str:
    escaped = term.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")
    return f"%{escaped}%"


def search_messages(
    conn: sqlite3.Connection, term: str, limit: int = 50
) -> list[sqlite3.Row]:
    term = term.strip()
    if not term:
        return []
    if len(term) >= 3:
        try:
            return conn.execute(
                "SELECT m.session_id, m.message_id, m.body FROM message_fts"
                " JOIN messages m ON m.session_id=message_fts.session_id"
                " AND m.message_id=message_fts.message_id"
                " WHERE message_fts MATCH ? LIMIT ?",
                (_fts_expr(term), limit),
            ).fetchall()
        except sqlite3.OperationalError:
            pass
    return conn.execute(
        "SELECT session_id, message_id, body FROM messages"
        " WHERE body LIKE ? ESCAPE '\\' LIMIT ?",
        (_like(term), limit),
    ).fetchall()


def list_sessions(
    conn: sqlite3.Connection,
    project_id: str | None = None,
    source_kind: str | None = None,
    date_from: str | None = None,
    date_to: str | None = None,
    q: str | None = None,
    limit: int | None = 100,
) -> list[dict]:
    clauses: list[str] = []
    params: list[object] = []
    if project_id:
        clauses.append("s.project_id = ?")
        params.append(project_id)
    if source_kind:
        clauses.append("s.source_kind = ?")
        params.append(source_kind)
    day = "substr(COALESCE(s.updated_at, s.created_at, ''), 1, 10)"
    if date_from:
        clauses.append(f"{day} >= ?")
        params.append(date_from)
    if date_to:
        clauses.append(f"{day} <= ?")
        params.append(date_to)
    if q:
        term = q.strip()
        if len(term) >= 3:
            clauses.append(
                "s.session_id IN (SELECT session_id FROM message_fts"
                " WHERE message_fts MATCH ?)"
            )
            params.append(_fts_expr(term))
        else:
            clauses.append(
                "(s.title LIKE ? ESCAPE '\\' OR s.summary LIKE ? ESCAPE '\\'"
                " OR s.session_id IN (SELECT session_id FROM messages"
                " WHERE body LIKE ? ESCAPE '\\'))"
            )
            params.extend([_like(term)] * 3)
    sql = (
        "SELECT s.*, p.display_name FROM sessions s"
        " LEFT JOIN projects p ON p.project_id = s.project_id"
    )
    if clauses:
        sql += " WHERE " + " AND ".join(clauses)
    sql += " ORDER BY COALESCE(s.updated_at, s.created_at, '') DESC"
    if limit is not None:
        sql += " LIMIT ?"
        params.append(limit)
    return [dict(row) for row in conn.execute(sql, params)]


def get_session(conn: sqlite3.Connection, session_id: str) -> dict | None:
    row = conn.execute(
        "SELECT s.*, p.display_name FROM sessions s"
        " LEFT JOIN projects p ON p.project_id = s.project_id"
        " WHERE s.session_id=?",
        (session_id,),
    ).fetchone()
    return dict(row) if row else None


def get_messages(conn: sqlite3.Connection, session_id: str) -> list[dict]:
    rows = conn.execute(
        "SELECT * FROM messages WHERE session_id=? ORDER BY seq, message_id",
        (session_id,),
    )
    return [dict(row) for row in rows]


def get_projects(conn: sqlite3.Connection) -> list[dict]:
    rows = conn.execute(
        "SELECT p.*, COUNT(s.session_id) AS session_count FROM projects p"
        " LEFT JOIN sessions s ON s.project_id = p.project_id"
        " GROUP BY p.project_id ORDER BY p.display_name"
    )
    return [dict(row) for row in rows]


def stats(conn: sqlite3.Connection) -> dict[str, int]:
    return {
        "projects": conn.execute("SELECT COUNT(*) FROM projects").fetchone()[0],
        "sessions": conn.execute("SELECT COUNT(*) FROM sessions").fetchone()[0],
        "messages": conn.execute("SELECT COUNT(*) FROM messages").fetchone()[0],
        "fts_rows": conn.execute("SELECT COUNT(*) FROM message_fts").fetchone()[0],
    }


def load_import_files(conn: sqlite3.Connection) -> dict[str, sqlite3.Row]:
    return {row["path"]: row
            for row in conn.execute("SELECT * FROM import_files")}
```

创建 `sessionpack/catalog/write.py`：

```python
"""catalog 的写入：项目/主题/消息 upsert、哈希刷新、导入进度表。

FTS 索引与 messages 表必须成对写入，故 upsert_message 内的
message_fts INSERT 与 messages INSERT 不可拆散。
"""

from __future__ import annotations

import hashlib
import json
import sqlite3
from typing import NamedTuple

from sessionpack.model import MessageRecord, ProjectRecord, TopicRecord


class WriteResult(NamedTuple):
    inserted: bool
    session_created: bool


def upsert_project(conn: sqlite3.Connection, project: ProjectRecord) -> None:
    conn.execute(
        "INSERT INTO projects(project_id, display_name, local_path_hint, source_kind)"
        " VALUES(?,?,?,?) ON CONFLICT(project_id) DO UPDATE SET"
        " display_name=excluded.display_name, local_path_hint=excluded.local_path_hint",
        (project.project_id, project.display_name,
         project.local_path_hint, project.source_kind),
    )


def upsert_topic(conn: sqlite3.Connection, rec: TopicRecord) -> WriteResult:
    existing = conn.execute(
        "SELECT updated_at FROM sessions WHERE session_id=?", (rec.session_id,)
    ).fetchone()
    created = existing is None
    title = rec.body[:60] or None
    if created:
        conn.execute(
            "INSERT INTO sessions(session_id, project_id, source_kind, title, summary,"
            " created_at, updated_at) VALUES(?,?,?,?,?,?,?)",
            (rec.session_id, rec.project_id, rec.source_kind, title, rec.body,
             rec.topic_time, rec.topic_time),
        )
        return WriteResult(True, True)
    if rec.topic_time >= (existing["updated_at"] or ""):
        conn.execute(
            "UPDATE sessions SET project_id=?, title=?, summary=?, updated_at=?"
            " WHERE session_id=?",
            (rec.project_id, title, rec.body, rec.topic_time, rec.session_id),
        )
    return WriteResult(True, False)


def upsert_message(conn: sqlite3.Connection, rec: MessageRecord) -> WriteResult:
    cursor = conn.execute(
        "INSERT OR IGNORE INTO messages(session_id, message_id, seq, ts, intent,"
        " outcome, actions, learned, meta_json, body) VALUES(?,?,?,?,?,?,?,?,?,?)",
        (rec.session_id, rec.message_id, rec.seq, rec.ts, rec.intent, rec.outcome,
         "|".join(rec.actions), "|".join(rec.learned),
         json.dumps(rec.meta, ensure_ascii=False) if rec.meta else None, rec.body),
    )
    if cursor.rowcount == 0:
        return WriteResult(False, False)
    conn.execute(
        "INSERT INTO message_fts(session_id, message_id, text) VALUES(?,?,?)",
        (rec.session_id, rec.message_id, rec.body),
    )
    existing = conn.execute(
        "SELECT 1 FROM sessions WHERE session_id=?", (rec.session_id,)
    ).fetchone()
    created = existing is None
    if created:
        conn.execute(
            "INSERT INTO sessions(session_id, project_id, source_kind, title,"
            " created_at, updated_at) VALUES(?,?,?,?,?,?)",
            (rec.session_id, rec.project_id, rec.source_kind,
             (rec.intent or rec.body)[:60] or None, rec.ts, rec.ts),
        )
    else:
        conn.execute(
            "UPDATE sessions SET project_id=?,"
            " updated_at=MAX(COALESCE(updated_at,''), COALESCE(?, ''))"
            " WHERE session_id=?",
            (rec.project_id, rec.ts, rec.session_id),
        )
    return WriteResult(True, created)


def refresh_session_hashes(conn: sqlite3.Connection, session_ids: set[str]) -> None:
    for session_id in sorted(session_ids):
        ids = [
            row[0]
            for row in conn.execute(
                "SELECT message_id FROM messages WHERE session_id=?"
                " ORDER BY seq, message_id",
                (session_id,),
            )
        ]
        digest = hashlib.sha256("\n".join(ids).encode("utf-8")).hexdigest()
        conn.execute(
            "UPDATE sessions SET content_hash=? WHERE session_id=?",
            (digest, session_id),
        )


# ---------------------------------------------------------------- run / file 表


def begin_import_run(conn: sqlite3.Connection, run_id: str, adapter_id: str,
                     adapter_version: int, started_at: str) -> None:
    conn.execute(
        "INSERT INTO import_runs(run_id, adapter_id, adapter_version,"
        " started_at, result) VALUES(?,?,?,?,?)",
        (run_id, adapter_id, adapter_version, started_at, "running"))


def finish_import_run(conn: sqlite3.Connection, run_id: str, *,
                      finished_at: str, result: str,
                      scanned_files: int | None = None,
                      imported_rows: int | None = None,
                      failed_rows: int | None = None,
                      warnings_json: str | None = None) -> None:
    cols: dict[str, object] = {
        "finished_at": finished_at, "result": result,
        "scanned_files": scanned_files, "imported_rows": imported_rows,
        "failed_rows": failed_rows, "warnings_json": warnings_json,
    }
    parts = ", ".join(f"{k}=?" for k, v in cols.items() if v is not None)
    args = [v for v in cols.values() if v is not None]
    conn.execute(f"UPDATE import_runs SET {parts} WHERE run_id=?",
                 (*args, run_id))


def register_import_files(conn: sqlite3.Connection,
                          rows: list[tuple[str, int, float]]) -> None:
    conn.executemany(
        "INSERT OR IGNORE INTO import_files(path, size, mtime) VALUES(?,?,?)",
        rows)


def reset_import_file(conn: sqlite3.Connection, *, size: int, mtime: float,
                      path: str) -> None:
    conn.execute(
        "UPDATE import_files SET size=?, mtime=?, offset=0, rows=0,"
        " done=0, error=NULL WHERE path=?", (size, mtime, path))


def clear_import_file_error(conn: sqlite3.Connection, path: str) -> None:
    conn.execute(
        "UPDATE import_files SET offset=0, rows=0, done=0, error=NULL"
        " WHERE path=?", (path,))


def save_import_file_state(conn: sqlite3.Connection, *, path: str, offset: int,
                           rows: int, done: int, last_run_id: str,
                           error: str | None) -> None:
    conn.execute(
        "UPDATE import_files SET offset=?, rows=?, done=?, last_run_id=?,"
        " error=? WHERE path=?",
```

```python
def require_catalog(path: Path) -> bool:
```

```python
"""cli 的渲染层：业务函数只产出 CommandResult，打印与退出码在此统一决定。

本模块在 lint_imports 中登记为 L0：只依赖标准库，不 import 任何 sessionpack
业务模块，因此任何命令模块都能安全依赖它。
"""

from __future__ import annotations

import json
import sys
from dataclasses import dataclass, field
from typing import Any


@dataclass
class CommandResult:
    """一条命令的渲染输入：退出码 + 文本行 + JSON 载荷。"""

    exit_code: int
    out_lines: list[str] = field(default_factory=list)
    err_lines: list[str] = field(default_factory=list)
    data: Any | None = None


def emit(res: CommandResult, as_json: bool) -> int:
    """渲染结果并返回退出码。

    as_json 且 data 非 None 时只输出 JSON；否则输出文本行（stdout）与错误行
    （stderr）。data 为 None 时 as_json 不生效，恒走文本/错误行。
    """
    if as_json and res.data is not None:
        print(json.dumps(res.data, ensure_ascii=False, indent=2))
    else:
        for line in res.out_lines:
            print(line)
        for line in res.err_lines:
            print(line, file=sys.stderr)
    return res.exit_code
```

```python
from __future__ import annotations

import sys
from pathlib import Path

from sessionpack import __version__, archive, diagnostics

def run_doctor(args) -> CommandResult:
    data = diagnostics.collect(memory_root(args), layout(args))
    code = EXIT_OK if data["memory_root_exists"] else EXIT_USAGE

        return CommandResult(code, data=data)
    lines = [
        f"SessionPack {__version__}",
        f"记忆目录：{data['memory_root']}"
        f"（{'存在' if data['memory_root_exists'] else '不存在'}）",

    source = data["source"]
    if source:
        lines.append(f"适配器：{source['adapter_id']} v{source['adapter_version']}")
        lines.append(f"项目 {source['projects']} 个（目录 {source['project_dirs']} 个）")

        if source["date_min"]:
            lines.append(f"日期范围：{source['date_min']} 至 {source['date_max']}")

    lines.append(f"私有库：{private['path']}"

    stats = catalog_info["stats"]
    lines.append(f"索引统计：项目 {stats['projects']}，会话 {stats['sessions']}，"
                 f"消息 {stats['messages']}，FTS {stats['fts_rows']}")
    return CommandResult(code, out_lines=lines)

def run_scan(args) -> CommandResult:
    root = memory_root(args)
    if not root.is_dir():
        return CommandResult(EXIT_USAGE, err_lines=[f"记忆目录不存在: {root}"])
    info = get_adapter().inspect(root)
    if args.json:
        return CommandResult(EXIT_OK, data=info.as_dict())
    return CommandResult(EXIT_OK, out_lines=[
        f"适配器：{info.adapter_id} v{info.adapter_version}",
        f"项目 {info.project_count} 个（目录 {info.project_dir_count} 个）",

    return emit(run_scan(args), args.json)


def run_verify(args) -> CommandResult:
    results = [archive.verify_archive(Path(raw)) for raw in args.archives]
    code = EXIT_OK if all(result.ok for result in results) else EXIT_VERIFY
    if args.json:
        return CommandResult(code, data=[
            {"path": str(result.path), "ok": result.ok,
             "issues": [{"code": issue.code, "detail": issue.detail}
                        for issue in result.issues]}
            for result in results
        ])
    lines: list[str] = []
    for result in results:
        if result.ok:
            assert result.manifest is not None  # verify_archive 成功时 manifest 必非空
            lines.append(f"OK  {result.path}"
                         f"（archive_id {result.manifest['archive_id']}）")
        else:
            lines.append(f"BAD {result.path}："
                         + ", ".join(issue.code for issue in result.issues))
    return CommandResult(code, out_lines=lines)

        return EXIT_USAGE
    from sessionpack.server import serve

    serve(lay.catalog, port=args.port, open_browser=not args.no_browser,
          memory_root=memory_root(args))
    return EXIT_OK
```

```python
"""cli 的写入型命令：import / export / push / pull。

业务编排集中在 run_xxx（不 print），cmd_xxx 只负责调用 emit 渲染并返回退出码。
"""

from __future__ import annotations

import sqlite3
from pathlib import Path

from sessionpack import archive, catalog, config, export, importer, sync
from sessionpack.adapters import get_adapter
from sessionpack.audit import ReadAudit, ReadBoundaryError
from sessionpack.cli.common import (
    EXIT_ERROR,
    EXIT_OK,
    EXIT_USAGE,
    EXIT_VERIFY,
    layout,
    memory_root,
    require_catalog,
    resolve_sync_dir,
)
from sessionpack.cli.output import CommandResult, emit


def run_import(args) -> CommandResult:
    lay = layout(args)
    if args.archive:
        conn = catalog.connect(lay.catalog)
        catalog.init_schema(conn)
        try:
            result = archive.import_archive(conn, Path(args.archive))
        except archive.ArchiveError as exc:
            return CommandResult(EXIT_VERIFY, err_lines=[
                "归档校验失败: " + ", ".join(issue.code for issue in exc.issues)])
        except sqlite3.Error as exc:
            return CommandResult(EXIT_ERROR, err_lines=[f"数据库错误: {exc}"])
        finally:
            conn.close()
        return CommandResult(EXIT_OK, out_lines=[
            f"归档 {result.archive_id}：会话 {result.sessions}，"
            f"消息 {result.messages}，新增消息 {result.messages_inserted}"])

    root = memory_root(args)
    if not root.is_dir():
        return CommandResult(EXIT_USAGE, err_lines=[f"记忆目录不存在: {root}"])
    adapter = get_adapter()
    declared = adapter.declared_paths(root)
    kwargs = dict(reparse=args.reparse, retry_failed=args.retry_failed,
                  project_filter=args.project,
                  since=args.since, until=args.until)
    boundary_lines: list[str] = []
    try:
        if args.audit:
            with ReadAudit(root, declared) as audit:
                stats = importer.import_source(adapter, root, lay.catalog, **kwargs)
            try:
                audit.check(declared)
            except ReadBoundaryError as exc:
                return CommandResult(EXIT_ERROR, err_lines=[str(exc)])
            if not args.json:
                boundary_lines.append(
                    f"读取边界：审计通过（打开 {len(audit.opened)} 个文件）")
        else:
            stats = importer.import_source(adapter, root, lay.catalog, **kwargs)
    except sqlite3.Error as exc:
        return CommandResult(EXIT_ERROR, err_lines=[f"数据库错误: {exc}"])

    if args.json:
        return CommandResult(EXIT_OK, data=stats.as_dict())
    lines = [
        f"导入完成：文件 {stats.files_processed}/{stats.files_declared}，"
        f"跳过 {stats.files_skipped}，消息 {stats.messages_inserted}，"
        f"主题 {stats.topics_applied}，失败 {stats.rows_failed} 行，"
        f"用时 {stats.elapsed_s:.2f}s"
    ]
    if stats.stopped:
        lines.append("已按停止条件中断，下次运行将自动续接。")
    return CommandResult(EXIT_OK, out_lines=boundary_lines + lines)


def cmd_import(args) -> int:
    return emit(run_import(args), args.json)


def run_export(args) -> CommandResult:
    lay = layout(args)
    if not require_catalog(lay.catalog):
        return CommandResult(EXIT_USAGE,
                             err_lines=["索引不存在，请先运行 sessionpack import"])
    conn = catalog.connect(lay.catalog)
    out = Path(args.out)
    try:
        if args.archive:
            if args.session:
                return CommandResult(
                    EXIT_USAGE,
                    err_lines=["归档不支持按会话生成，请用 --project 或 --all"])
            project_ids = ([args.project] if args.project
                           else [row["project_id"]
                                 for row in catalog.get_projects(conn)])
            lines = []
            for project_id in project_ids:
                item = archive.build_archive(conn, out, project_id)
                lines.append(f"已生成 {item.path}（archive_id {item.archive_id}，"
                             f"会话 {item.sessions}，消息 {item.messages}）")
            return CommandResult(EXIT_OK, out_lines=lines)
        if args.session:
            try:
                path, _ = export.export_session(conn, args.session, out, args.format)
            except KeyError:
                return CommandResult(EXIT_USAGE,
                                     err_lines=[f"会话不存在: {args.session}"])
            return CommandResult(EXIT_OK, out_lines=[f"已导出 {path}"])
        if args.project:
            if not catalog.list_sessions(conn, project_id=args.project, limit=1):
                return CommandResult(
                    EXIT_USAGE, err_lines=[f"项目不存在或无会话: {args.project}"])
            result = export.export_project(conn, args.project, out, args.format)
        else:
            result = export.export_all(conn, out, args.format)
        return CommandResult(EXIT_OK, out_lines=[
            f"已导出 {result.sessions} 个会话、{result.messages} 条消息到 {out}"])
    except ValueError as exc:
        return CommandResult(EXIT_ERROR, err_lines=[f"导出失败: {exc}"])
    except sqlite3.Error as exc:
        return CommandResult(EXIT_ERROR, err_lines=[f"数据库错误: {exc}"])
    finally:
        conn.close()


def cmd_export(args) -> int:
    # export 没有 --json 参数，恒走文本渲染
    return emit(run_export(args), False)


def run_push(args) -> CommandResult:
    lay = layout(args)
    target = resolve_sync_dir(args, lay)
    if target is None:
        return CommandResult(
            EXIT_USAGE,
            err_lines=["未指定同步目录：请加 --dir DIR，"
                       "或先运行 sessionpack push --dir DIR 落配置"])
    result = sync.push_archives(lay.archives, target, dry_run=args.dry_run)
    payload = {
        "sync_dir": str(target),
        "pushed": result.pushed,
        "skipped_existing": result.skipped_existing,
        "skipped_unverified": list(result.skipped_unverified),
        "bytes": result.bytes,
        "errors": list(result.errors),
    }
    prefix = "[dry-run] " if args.dry_run else ""
    verb = "将推送" if args.dry_run else "已推送"
    lines = [f"{prefix}{verb} {result.pushed} 个包共 {result.bytes} 字节，"
             f"跳过已存在 {result.skipped_existing}，"
             f"跳过未通过校验 {len(result.skipped_unverified)}"]
    lines += [f"跳过未通过校验的包: {item}" for item in result.skipped_unverified]
    errors = [f"推送失败: {item}" for item in result.errors]
    if result.errors:
        return CommandResult(EXIT_ERROR, out_lines=lines, err_lines=errors, data=payload)
    # 落配置对 JSON 与文本两种模式同样生效，与原文一致
    if args.dir and not args.dry_run:
        config.set_sync_dir(lay.root, target)
    return CommandResult(EXIT_OK, out_lines=lines, data=payload)


def cmd_push(args) -> int:
    return emit(run_push(args), args.json)


def run_pull(args) -> CommandResult:
    lay = layout(args)
    target = resolve_sync_dir(args, lay)
    if target is None:
        return CommandResult(
            EXIT_USAGE,
            err_lines=["未指定同步目录：请加 --dir DIR，"
                       "或先运行 sessionpack push --dir DIR 落配置"])
    if not target.is_dir():
        return CommandResult(EXIT_USAGE, err_lines=[f"同步目录不存在: {target}"])
    conn = catalog.connect(lay.catalog)
    catalog.init_schema(conn)
    try:
        result = sync.pull_archives(lay.archives, target, conn)
    finally:
        conn.close()
    if args.json:
        return CommandResult(EXIT_ERROR if result.errors else EXIT_OK, data={
            "sync_dir": str(target),
            "pulled": result.pulled,
            "skipped_existing": result.skipped_existing,
            "skipped_invalid": list(result.skipped_invalid),
            "sessions_added": result.sessions_added,
            "messages_added": result.messages_added,
            "bytes": result.bytes,
            "errors": list(result.errors),
        })
    lines = [f"已拉取 {result.pulled} 个包（新增会话 {result.sessions_added}、"
             f"消息 {result.messages_added}），跳过已存在 {result.skipped_existing}，"
             f"跳过无效包 {len(result.skipped_invalid)}"]
    lines += [f"跳过无效包: {item}" for item in result.skipped_invalid]
    if result.errors:
        return CommandResult(EXIT_ERROR, out_lines=lines,
                             err_lines=[f"拉取失败: {item}" for item in result.errors])
    return CommandResult(EXIT_OK, out_lines=lines)


def cmd_pull(args) -> int:
    return emit(run_pull(args), args.json)
```

```python
def sql_violations(tree: ast.AST, rel: str) -> list[str]:
    """SQL 独占令：字符串字面量含受管表写语句，仅 catalog 包允许。

    docstring 只是说明性文字，先收集其节点 id 再跳过，避免「注释里提到
    INSERT INTO」被判成违规。
    """
    if rel.startswith("sessionpack/catalog/"):
        return []
    docstrings = {
        id(node.value)
        for node in ast.walk(tree)
        if isinstance(node, (ast.Module, ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef))
        and node.body and isinstance(node.body[0], ast.Expr)
        and isinstance(node.body[0].value, ast.Constant)
        and isinstance(node.body[0].value.value, str)
    }
    out: list[str] = []
    for node in ast.walk(tree):
        if (isinstance(node, ast.Constant) and isinstance(node.value, str)
                and id(node) not in docstrings
                and SQL_WRITE.search(node.value)):
            out.append(f"{rel}:{node.lineno}: E_SQL raw SQL write on "
                       "managed table (only catalog package allowed)")
```
