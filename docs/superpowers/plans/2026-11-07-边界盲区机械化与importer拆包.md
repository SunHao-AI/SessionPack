```python
def _rebound_names(tree: ast.AST) -> set[str]:
    """文件内被重新绑定过的名字（赋值/增广赋值/带注解赋值/海象/for/with/
    except-as 目标、函数参数）。属性链根名命中即保守放行（Spec 4 §3.2）：
    同名实例会误报为模块私有访问，无法静态断定时宁可漏报不误报。
    取舍在案：'先 import 模块、后同名重绑定、再跨模块访问私有属性'的
    真实违例会被漏掉——该模式极罕见且本身是坏味道，接受。"""
    names: set[str] = set()

    def _add(t: ast.expr | None) -> None:
        if isinstance(t, ast.Name):
            names.add(t.id)
        elif isinstance(t, (ast.Tuple, ast.List)):
            for elt in t.elts:
                _add(elt)
        elif isinstance(t, ast.Starred):
            _add(t.value)

    for node in ast.walk(tree):
        if isinstance(node, ast.Assign):
            for t in node.targets:
                _add(t)
        elif isinstance(node, (ast.AugAssign, ast.AnnAssign, ast.NamedExpr)):
            _add(node.target)
        elif isinstance(node, (ast.For, ast.AsyncFor)):
            _add(node.target)
        elif isinstance(node, ast.withitem):
            _add(node.optional_vars)
        elif isinstance(node, ast.ExceptHandler):
            if node.name:
                names.add(node.name)
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            a = node.args
            for arg in [*a.posonlyargs, *a.args, *a.kwonlyargs,
                        a.vararg, a.kwarg]:
                if arg is not None:
                    names.add(arg.arg)
    return names
```

注意：**不**把 `import x as y` 的别名计入 rebound——import 别名是模块引用本身，计入会把既有"别名式私有访问"检查整个放空（spec §3.2 明确该路不受收紧影响）。
